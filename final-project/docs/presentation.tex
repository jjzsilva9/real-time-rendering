\begin{frame}
\frametitle{Technical Implementation}

\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \textbf{Following the paper}
        \begin{itemize}
            \item CPU SVO construction from triangle mesh
            \item Radiance injection via light-view rasterisation
            \item Bottom-up Gaussian MIP-filtering with neighbour pointers
            \item Diffuse \& specular cone tracing at final gather
            \item Phong-derived specular cone aperture ($\psi \approx \sqrt{2/N_s}$)
            \item Shadow map for direct illumination
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Deviations}
        \begin{itemize}
            \item Voxelisation on CPU (paper uses GPU rasterisation pipeline)
            \item Isotropic voxels only (paper proposes anisotropic directional pre-integration)
            \item No brick/tile texture storage; single linear SSBO pool
            \item Dynamic objects cast direct shadows but are not re-voxelised per-frame; no separate dynamic SVO region means full injection \& filtering runs every frame regardless
        \end{itemize}
    \end{column}
\end{columns}

\end{frame}

\begin{frame}
\frametitle{Results \& Evaluation}

\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \textbf{Visual quality}
        \begin{itemize}
            \item Plausible indirect diffuse with colour bleeding
            \item Glossy reflections via specular cone, aperture tied to $N_s$
            \item Smooth GI transitions via Gaussian MIP-filtering
            \item Direct shadows from static \& dynamic geometry
        \end{itemize}
        \vspace{0.4em}
        \textbf{Performance (128\textsuperscript{3}, GTX series)}
        \begin{tabular}{lr}
            \toprule
            Configuration & FPS \\
            \midrule
            Static scene & $\sim$60 \\
            + Dynamic objects & $\sim$10 \\
            \bottomrule
        \end{tabular}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Limitations}
        \begin{itemize}
            \item No anisotropic voxels — some light leaking at large cone angles
            \item Dynamic objects invisible to GI; full re-injection cost every frame
            \item Single-bounce diffuse only (specular provides approximate second bounce)
            \item SVO built on CPU — rebuild on resolution change is slow
        \end{itemize}
    \end{column}
\end{columns}

\end{frame}
